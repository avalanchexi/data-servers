-- 客户授信额度查询:支持按客户号、产品类型查询当前可用额度
SELECT
    a.cust_no,
    a.cust_name,
    a.product_type,
    a.credit_limit,
    a.used_limit,
    a.available_limit,
    b.risk_level
FROM ads_credit_limit a
LEFT JOIN ads_credit_dimension b
    ON a.cust_no = b.cust_no
WHERE
    a.cust_no = ${custNo}
    AND a.product_type = ${productType}
    AND a.status = 'ACTIVE'
ORDER BY a.credit_limit DESC
-- 系统自动识别,无需分号结尾
